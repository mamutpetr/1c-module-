           И (КэшСостоянияП2.НН = ННСсылка)

Функция СловаПохожи(Слово1, Слово2)

    Перем ДлинаОбщая, ДлинаПроверки, РазницаДлин;

    Если Слово1 = Слово2 Тогда
        Возврат 1;
    КонецЕсли;
    Если (СтрДлина(Слово1) < 5) Или (СтрДлина(Слово2) < 5) Тогда
        Возврат 0;
    КонецЕсли;

    ДлинаОбщая = СтрДлина(Слово1);
    Если СтрДлина(Слово2) < ДлинаОбщая Тогда
        ДлинаОбщая = СтрДлина(Слово2);
    КонецЕсли;
    РазницаДлин = СтрДлина(Слово1) - СтрДлина(Слово2);
    Если РазницаДлин < 0 Тогда
        РазницаДлин = -РазницаДлин;
    КонецЕсли;
    Если РазницаДлин > 3 Тогда
        Возврат 0;
    КонецЕсли;

    // Допускаем одну-две ошибки OCR: например, Броколі/Брокколі.
    ДлинаПроверки = ДлинаОбщая - 2;
    Если ДлинаПроверки > 4 Тогда
        ДлинаПроверки = 4;
    КонецЕсли;

    Если Лев(Слово1, ДлинаПроверки) = Лев(Слово2, ДлинаПроверки) Тогда
        Возврат 1;
    КонецЕсли;
    Возврат 0;

Функция ЕстьПохожееСлово(Слово, НормИмя)

    Перем РабочаяСтрока, Позиция, СловоКандидат;

    РабочаяСтрока = СокрЛП(НормИмя);
    Пока ПустаяСтрока(РабочаяСтрока) = 0 Цикл
        Позиция = Найти(РабочаяСтрока, " ");
        Если Позиция = 0 Тогда
            СловоКандидат = РабочаяСтрока;
            РабочаяСтрока = "";
        Иначе
            СловоКандидат = Лев(РабочаяСтрока, Позиция - 1);
            РабочаяСтрока = СокрЛП(Сред(РабочаяСтрока, Позиция + 1, СтрДлина(РабочаяСтрока) - Позиция));
        КонецЕсли;

        Если СловаПохожи(Слово, СловоКандидат) = 1 Тогда
            Возврат 1;
        КонецЕсли;
    КонецЦикла;

    Возврат 0;

Функция ЦенаПохожа(ЦенаСкана, ЦенаБезНДС, СуммаНДС, Количество)

    Перем Разница, ЦенаСНДС;

    Если (ЦенаСкана <= 0) Или (ЦенаБезНДС <= 0) Тогда
        Возврат 0;
    КонецЕсли;

    Разница = ЦенаСкана - ЦенаБезНДС;
    Если Разница < 0 Тогда
        Разница = -Разница;
    КонецЕсли;
    Если Разница / ЦенаБезНДС <= 0.35 Тогда
        Возврат 1;
    КонецЕсли;

    Если Количество > 0 Тогда
        ЦенаСНДС = ЦенаБезНДС + СуммаНДС / Количество;
        Разница = ЦенаСкана - ЦенаСНДС;
        Если Разница < 0 Тогда
            Разница = -Разница;
        КонецЕсли;
        Если Разница / ЦенаСНДС <= 0.15 Тогда
            Возврат 1;
        КонецЕсли;
    КонецЕсли;

    Возврат 0;

Функция ОценкаЦены(ЦенаСкана, ЦенаБезНДС, СуммаНДС, Количество)

    Перем ЦенаСНДС, Разница, Относительная, ЛучшаяОценка;

    ЛучшаяОценка = 0;
    Если (ЦенаСкана <= 0) Тогда
        Возврат 0;
    КонецЕсли;

    // Скан може містити або ціну без ПДВ, або ціну з ПДВ.
    Если ЦенаБезНДС > 0 Тогда
        Разница = ЦенаСкана - ЦенаБезНДС;
        Если Разница < 0 Тогда
            Разница = -Разница;
        КонецЕсли;
        Относительная = Разница / ЦенаБезНДС;
        Если Относительная <= 0.01 Тогда
            ЛучшаяОценка = 120;
        ИначеЕсли Относительная <= 0.05 Тогда
            ЛучшаяОценка = 80;
        ИначеЕсли Относительная <= 0.15 Тогда
            ЛучшаяОценка = 40;
        ИначеЕсли Относительная <= 0.35 Тогда
            ЛучшаяОценка = 10;
        КонецЕсли;
    КонецЕсли;

    Если (Количество > 0) И (ЦенаБезНДС > 0) Тогда
        ЦенаСНДС = ЦенаБезНДС + СуммаНДС / Количество;
        Разница = ЦенаСкана - ЦенаСНДС;
        Если Разница < 0 Тогда
            Разница = -Разница;
        КонецЕсли;
        Относительная = Разница / ЦенаСНДС;
        Если Относительная <= 0.01 Тогда
            Если ЛучшаяОценка < 120 Тогда
                ЛучшаяОценка = 120;
            КонецЕсли;
        ИначеЕсли Относительная <= 0.05 Тогда
            Если ЛучшаяОценка < 80 Тогда
                ЛучшаяОценка = 80;
            КонецЕсли;
        ИначеЕсли Относительная <= 0.15 Тогда
            Если ЛучшаяОценка < 40 Тогда
                ЛучшаяОценка = 40;
            КонецЕсли;
        ИначеЕсли Относительная <= 0.35 Тогда
            Если ЛучшаяОценка < 10 Тогда
                ЛучшаяОценка = 10;
            КонецЕсли;
        КонецЕсли;
    КонецЕсли;

    Возврат ЛучшаяОценка;

    // назву «... з куркою ...». Розводимо ці дві позиції до загального
    // підрахунку слів, щоб вони не давали помилку «кілька схожих товарів».
    Если (Найти(НормСкана, "млинці") > 0) И (Найти(НормСкана, "француз") > 0) Тогда
        Если Найти(НормСкана, "сир") > 0 Тогда
            Если Найти(Норм1С, "кур") > 0 Тогда
                Возврат 0;
            КонецЕсли;
            Если (Найти(Норм1С, "млинці") > 0) И (Найти(Норм1С, "француз") > 0) Тогда
                Возврат 900 + ОценкаЦены(ЦенаСкана, ЦенаБезНДС, СуммаНДС, Количество);
            КонецЕсли;
        ИначеЕсли Найти(НормСкана, "кур") > 0 Тогда
            Если Найти(Норм1С, "кур") = 0 Тогда
                Возврат 0;
            КонецЕсли;
            Если (Найти(Норм1С, "млинці") > 0) И (Найти(Норм1С, "француз") > 0) Тогда
                Возврат 900 + ОценкаЦены(ЦенаСкана, ЦенаБезНДС, СуммаНДС, Количество);
            КонецЕсли;
        КонецЕсли;
    КонецЕсли;

    Если НормСкана = Норм1С Тогда
        Возврат 1000;
    КонецЕсли;
    Если (СтрДлина(НормСкана) >= 8) И ((Найти(Норм1С, НормСкана) > 0) Или (Найти(НормСкана, Норм1С) > 0)) Тогда
        Возврат 800;
    КонецЕсли;

    Если (СтрДлина(ЧислаСкана) > 1) И (СтрДлина(Числа1С) > 1) Тогда
        Если ЕстьОбщееЧисло(ЧислаСкана, Числа1С) = 0 Тогда
            Возврат 0;
        КонецЕсли;
    КонецЕсли;

    СловСовпало = КоличествоСовпавшихСловНорм(НормСкана, Норм1С);
    ТочныхСлов = КоличествоТочныхСловНорм(НормСкана, Норм1С);
    Если СловСовпало >= 2 Тогда
        Результат = 70 + СловСовпало * 20;
    ИначеЕсли (СловСовпало = 1) И (ЕстьОбщееЧисло(ЧислаСкана, Числа1С) = 1) И (ЕстьДлинноеОбщееСловоНорм(НормСкана, Норм1С) = 1) Тогда
        Результат = 70;
    Иначе
        Возврат 0;
    КонецЕсли;

    // Точное длинное слово (например, «семикомпонентна» или «філейні»)
    // сильнее повышает оценку, чем приблизительное совпадение общего слова.
    Если ТочныхСлов >= 2 Тогда
        Результат = Результат + 100;
    ИначеЕсли ТочныхСлов = 1 Тогда
        Результат = Результат + 60;
    КонецЕсли;

    Результат = Результат + ОценкаЦены(ЦенаСкана, ЦенаБезНДС, СуммаНДС, Количество);

    Возврат Результат;

Функция ПроверитьТовары(Источник, НН, Товары)

    Перем ИмяТовара, Количество, ЛучшаяОценка, ВтораяОценка, Оценка;
    Перем ЛучшийТМЦ, ЛучшаяКво, НашлиВНН, ЛучшийКлючТМЦ, ВторойКлючТМЦ;

    Товары.ВыбратьСтроки();
    Пока Товары.ПолучитьСтроку() = 1 Цикл
        ИмяТовара = Товары.Наименование;
        Количество = Товары.Кво;
        Если (ПустаяСтрока(ИмяТовара) = 1) Или (Количество <= 0) Тогда
            Возврат "В строке импорта отсутствует товар или количество";
        КонецЕсли;

        // Партия и дата уже выбрали конкретную РасходнуюНакладную.
        // Ищем только среди ее строк: сначала точное имя, затем ключевые слова,
        // вес из названия и, как дополнительный признак, цену.
        ЛучшаяОценка = 0;
        ВтораяОценка = 0;
        ЛучшаяКво = 0;
        ЛучшийКлючТМЦ = "";
        ВторойКлючТМЦ = "";
        Источник.ВыбратьСтроки();
        Пока Источник.ПолучитьСтроку() = 1 Цикл
            Оценка = ОценкаСоответствияТовара(ИмяТовара, Товары.ЦенаСкана, Источник.ТМЦ, Источник.ЦенаБезНДС, Источник.НДС, Источник.Кво);
            Если Оценка > ЛучшаяОценка Тогда
                ВтораяОценка = ЛучшаяОценка;
                ВторойКлючТМЦ = ЛучшийКлючТМЦ;
                ЛучшаяОценка = Оценка;
                ЛучшийТМЦ = Источник.ТМЦ;
                ЛучшийКлючТМЦ = Строка(Источник.ТМЦ);
                ЛучшаяКво = Источник.Кво;
            ИначеЕсли Оценка > ВтораяОценка Тогда
                ВтораяОценка = Оценка;
                ВторойКлючТМЦ = Строка(Источник.ТМЦ);
            КонецЕсли;
        КонецЦикла;
        Если ЛучшаяОценка < 70 Тогда
            Возврат "Товар не найден в РасходнаяНакладная: " + ИмяТовара;
        КонецЕсли;
        Если (ВтораяОценка > 0) И (ВторойКлючТМЦ <> ЛучшийКлючТМЦ) И ((ЛучшаяОценка - ВтораяОценка) < 15) Тогда
            Возврат "Найдено несколько похожих товаров в РасходнаяНакладная: " + ИмяТовара + ". Уточните название или цену";
        КонецЕсли;
        Если Количество > ЛучшаяКво Тогда
            Возврат "Количество возврата больше первичной отгрузки: " + ИмяТовара;
        КонецЕсли;

        // Сохраняем точную ссылку ТМЦ. Дальше документы создаются только по ней,
        // поэтому второй раз не пытаемся угадать товар по тексту.
        Товары.ТМЦ1С = ЛучшийТМЦ;

        НашлиВНН = 0;
        НН.ВыбратьСтроки();
        Пока НН.ПолучитьСтроку() = 1 Цикл
            Если НН.ТМЦ = ЛучшийТМЦ Тогда
                НашлиВНН = 1;
                Прервать;
            КонецЕсли;
        КонецЦикла;
        Если НашлиВНН = 0 Тогда
            Возврат "Товар не найден в первичной НалоговаяНакладная: " + ИмяТовара;
        КонецЕсли;
    КонецЦикла;

    Возврат "";

        Возвратная.ТМЦ,
        Возвратная.Партия
    );

    ИтСумма = 0;
    ИтКво = 0;
    ИтСеб = 0;
    фНашли = 0;

    НомерПроводок = ПроводкиЗагрузить(Источник);
    Перв = ПроводкиРН.ПолучитьЗначение(НомерПроводок, "Перв");
    Посл = Перв + ПроводкиРН.ПолучитьЗначение(НомерПроводок, "Кол") - 1;

    СчетТМЦ = Возвратная.ТМЦ.Счет;
    ТМЦВозврата = Возвратная.ТМЦ;
    ПартияВозврата = Возвратная.Партия;

    Для К = Перв По Посл Цикл

        Если Лев(ПроводкиСтроки.ПолучитьЗначение(К, "КтКод"), 1) = "2" Тогда
            ИтСеб = ИтСеб + ПроводкиСтроки.ПолучитьЗначение(К, "Сумма");
        КонецЕсли;

        Если ПроводкиСтроки.ПолучитьЗначение(К, "КтСчет") <> СчетТМЦ Тогда
            Продолжить;
        КонецЕсли;

        Если ПроводкиСтроки.ПолучитьЗначение(К, "Суб2") <> ТМЦВозврата Тогда
            Продолжить;
        КонецЕсли;

        Если ПартияВыбрана = 1 Тогда
            Если ПроводкиСтроки.ПолучитьЗначение(К, "Суб3") <> ПартияВозврата Тогда
                Продолжить;
            КонецЕсли;
        КонецЕсли;

        фНашли = 1;
        ИтСумма = ИтСумма + ПроводкиСтроки.ПолучитьЗначение(К, "Сумма");
        ИтКво = ИтКво + ПроводкиСтроки.ПолучитьЗначение(К, "Количество");

    КонецЦикла;

    Если фНашли <> 1 Тогда
        глКомментарий("Указанная партия не списывалась в выбранной расходной накладной!", 1, , "!!");
        Возврат;
    КонецЕсли;

    Возвратная.СуммаУчБезНДС = ?(
        ИтКво = 0,
        0,
        ИтСумма * (Возвратная.Кво * Возвратная.Коэффициент / ИтКво)
    );
    Возвратная.ЦенаУчБезНДС = ?(ИтКво = 0, 0, ИтСумма / ИтКво);

    // Налоговый учет — также повторяем штатную ИзмДокПродажи().
    СчетТМЦ_НУ = глПолучитьСчетНУ(Возвратная.ТМЦ.Счет);

    Если СчетТМЦ_НУ.Выбран() = 0 Тогда
        Возврат;
    КонецЕсли;

    ИтСуммаНУ = 0;
    ИтСуммаДоходаНУ = 0;

    Для К = Перв По Посл Цикл

        Если Лев(ПроводкиСтроки.ПолучитьЗначение(К, "КтКод"), 2) = "Н7" Тогда
            ИтСуммаДоходаНУ = ИтСуммаДоходаНУ + ПроводкиСтроки.ПолучитьЗначение(К, "Сумма");
        КонецЕсли;

        Если ПроводкиСтроки.ПолучитьЗначение(К, "КтСчет") <> СчетТМЦ_НУ Тогда
            Продолжить;
        КонецЕсли;

        Если ПроводкиСтроки.ПолучитьЗначение(К, "Суб2") <> ТМЦВозврата Тогда
            Продолжить;
        КонецЕсли;

        Если ПартияВыбрана = 1 Тогда
            Если ПроводкиСтроки.ПолучитьЗначение(К, "Суб3") <> ПартияВозврата Тогда
                Продолжить;
            КонецЕсли;
        КонецЕсли;

        ИтСуммаНУ = ИтСуммаНУ + ПроводкиСтроки.ПолучитьЗначение(К, "Сумма");

    КонецЦикла;

    Возвратная.СуммаУчБезНДС_НУ = ?(

        ИтСуммаНУ * (Возвратная.Кво * Возвратная.Коэффициент / ИтКво)

// v52: тело прежней P281ПрочитатьПартииИзРН без изменений алгоритма,
// только проводки берутся из кэша ПроводкиЗагрузить().
Процедура P281ПрочитатьПартииИзПроводок(Источник, ИскомыйТМЦ, Результат, ОшибкаПартии)

    Перем Партия, НомерСтроки, Счет281, НомерПроводок, Перв, Посл, К;

    Счет281 = СчетПоКоду("281");
    НомерПроводок = ПроводкиЗагрузить(Источник);
    Перв = ПроводкиРН.ПолучитьЗначение(НомерПроводок, "Перв");
    Посл = Перв + ПроводкиРН.ПолучитьЗначение(НомерПроводок, "Кол") - 1;

    Для К = Перв По Посл Цикл

        Если ПроводкиСтроки.ПолучитьЗначение(К, "КтСчет") <> Счет281 Тогда
            Продолжить;
        КонецЕсли;

        Если ПроводкиСтроки.ПолучитьЗначение(К, "Суб2") <> ИскомыйТМЦ Тогда
            Продолжить;
        КонецЕсли;

        Партия = ПроводкиСтроки.ПолучитьЗначение(К, "Суб3");
        Если P281ПартияНедопустима(Партия) = 1 Тогда
            ОшибкаПартии = "В Кт 281 РасходнойНакладной " + Источник.НомерДок
                + " у товара " + Строка(ИскомыйТМЦ) + " пустая или служебная партия";
            Возврат;
        КонецЕсли;

        НомерСтроки = P281НайтиСтрокуПартии(Результат, Партия);
        Если НомерСтроки = 0 Тогда
            Результат.НоваяСтрока();
            Результат.Партия = Партия;
            Результат.КвоБаза = 0;
        Иначе
            Результат.ПолучитьСтрокуПоНомеру(НомерСтроки);
        КонецЕсли;

        Результат.КвоБаза = Результат.КвоБаза + ПроводкиСтроки.ПолучитьЗначение(К, "Количество");

    КонецЦикла;

    Если Результат.КоличествоСтрок() = 0 Тогда
        ОшибкаПартии = "В проводках Кт 281 РасходнойНакладной " + Источник.НомерДок
            + " не найдено списание товара " + Строка(ИскомыйТМЦ);
        Возврат;
    КонецЕсли;

    Для НомерСтроки = 1 По Результат.КоличествоСтрок() Цикл
        Результат.ПолучитьСтрокуПоНомеру(НомерСтроки);
        Если Результат.КвоБаза <= 0 Тогда
            ОшибкаПартии = "Некорректное количество Кт 281 по партии " + Строка(Результат.Партия);
            Возврат;
        КонецЕсли;
    КонецЦикла;

    // Штатная форма ВозвратнаяНакладная распределяет несколько партий
    // в порядке Партия*. Держим тот же детерминированный порядок.
    Результат.Сортировать("Партия*");

Функция P281ПрочитатьПартииИзРН(Источник, ИскомыйТМЦ, ОшибкаПартии)

    Перем Результат, Ссылка, Н, К, Перв, Посл;

    ОшибкаПартии = "";
    Результат = СоздатьОбъект("ТаблицаЗначений");
    Результат.НоваяКолонка("Партия");
    Результат.НоваяКолонка("КвоБаза");

    Если Источник.Проведен() <> 1 Тогда
        ОшибкаПартии = "Исходная РасходнаяНакладная " + Источник.НомерДок + " не проведена";
        Возврат Результат;
    КонецЕсли;

    Ссылка = Источник.ТекущийДокумент();

    // v52: повторный запрос той же пары РН + ТМЦ (проверка плана, затем создание
    // возврата) берется из кэша файла: те же строки в том же порядке.
    Если КэшСостоянияП2Активен = 1 Тогда
        Для Н = 1 По ПартииР281Ключи.КоличествоСтрок() Цикл
            Если ПартииР281Ключи.ПолучитьЗначение(Н, "ТМЦ") <> ИскомыйТМЦ Тогда
                Продолжить;
            КонецЕсли;
            Если ПартииР281Ключи.ПолучитьЗначение(Н, "РН") <> Ссылка Тогда
                Продолжить;
            КонецЕсли;
            ОшибкаПартии = ПартииР281Ключи.ПолучитьЗначение(Н, "Ошибка");
            Перв = ПартииР281Ключи.ПолучитьЗначение(Н, "Перв");
            Посл = Перв + ПартииР281Ключи.ПолучитьЗначение(Н, "Кол") - 1;
            Для К = Перв По Посл Цикл
                Результат.НоваяСтрока();
                Результат.Партия = ПартииР281Строки.ПолучитьЗначение(К, "Партия");
                Результат.КвоБаза = ПартииР281Строки.ПолучитьЗначение(К, "КвоБаза");
            КонецЦикла;
            Возврат Результат;
        КонецЦикла;
    КонецЕсли;

    P281ПрочитатьПартииИзПроводок(Источник, ИскомыйТМЦ, Результат, ОшибкаПартии);

    Если КэшСостоянияП2Активен = 1 Тогда
        Перв = ПартииР281Строки.КоличествоСтрок() + 1;
        Для К = 1 По Результат.КоличествоСтрок() Цикл
            Результат.ПолучитьСтрокуПоНомеру(К);
            ПартииР281Строки.НоваяСтрока();
            ПартииР281Строки.Партия = Результат.Партия;
            ПартииР281Строки.КвоБаза = Результат.КвоБаза;
        КонецЦикла;
        ПартииР281Ключи.НоваяСтрока();
        ПартииР281Ключи.РН = Ссылка;
        ПартииР281Ключи.ТМЦ = ИскомыйТМЦ;
        ПартииР281Ключи.Ошибка = ОшибкаПартии;
        ПартииР281Ключи.Перв = Перв;
        ПартииР281Ключи.Кол = Результат.КоличествоСтрок();
    КонецЕсли;

    Возврат Результат;

Процедура ОпределитьТМЦВИсточнике(Источник, НН, ИмяТовара, ЦенаСкана, ЛучшийТМЦ, ОшибкаПоиска)

    НормСкана = НормализоватьИмяТовара(ИмяТовара);

        Если Оценка > ЛучшаяОценка Тогда

            ВторойКлючТМЦ = ЛучшийКлючТМЦ;

            ЛучшийТМЦ = ИменаРН.ТМЦ;
            ЛучшийКлючТМЦ = ИменаРН.Ключ;
        ИначеЕсли Оценка > ВтораяОценка Тогда

                    Разница = -Разница;
                КонецЕсли;
                ТекущаяОценка = 100000 - Разница * 100000;
            Иначе
                ТекущаяОценка = ОценкаЦены(ЦенаСкана, Источник.ЦенаБезНДС, Источник.НДС, Источник.Кво);
            КонецЕсли;

            Если (Нашли = 0) Или (ТекущаяОценка > ЛучшаяОценка) Тогда
                Нашли = 1;
                ПартияСтроки = Источник.Партия;
                КвоСтроки = Источник.Кво;
                ЦенаБезНДССтроки = Источник.ЦенаБезНДС;
                НДССтроки = Источник.НДС;
                СуммаСНДССтроки = Источник.СуммаСНДС;
                ОценкаЦеныСтроки = ТекущаяОценка;
                ЛучшаяОценка = ТекущаяОценка;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;

Функция P281РаспределитьСтроку(Источник, НН, ИскомыйТМЦ, ПартияПлана, КвоВозврата, ОшибкаПартии)

    Перем Результат, Партии, Нашли;
    Перем ИсходноеКво, Коэффициент, Единица, ЦенаБезНДС, ИсходнаяСуммаБезНДС, ИсходныйНДС;
    Перем ТекущийОстаток, УжеВозвращено, ПропуститьБаза, НужноБаза;
    Перем Н, ДоступноБаза, БеремБаза, КвоЧасти;

    ОшибкаПартии = "";

    Результат = СоздатьОбъект("ТаблицаЗначений");
    Результат.НоваяКолонка("Партия");
    Результат.НоваяКолонка("Кво");
    Результат.НоваяКолонка("Коэффициент");
    Результат.НоваяКолонка("Ед");
    Результат.НоваяКолонка("ЦенаБезНДС");
    Результат.НоваяКолонка("ИсходнаяСуммаБезНДС");
    Результат.НоваяКолонка("ИсходныйНДС");

    P281ПолучитьКоммерческуюСтроку(
        Источник, ИскомыйТМЦ, ПартияПлана,
        Нашли, ИсходноеКво, Коэффициент, Единица, ЦенаБезНДС, ИсходнаяСуммаБезНДС, ИсходныйНДС
    );

    Если Нашли = 0 Тогда
        ОшибкаПартии = "Не найдена исходная строка товара " + Строка(ИскомыйТМЦ)
            + " в РасходнойНакладной " + Источник.НомерДок;
        Возврат Результат;
    КонецЕсли;

    Если (КвоВозврата <= 0) Или (Коэффициент <= 0) Или (ИсходноеКво <= 0) Тогда
        ОшибкаПартии = "Некорректное количество или коэффициент товара " + Строка(ИскомыйТМЦ);
        Возврат Результат;
    КонецЕсли;

    // Используем тот же остаток по цепочке П2, что уже использует рабочий
    // DONOR PLANNER. Поэтому партийное исправление не меняет его бизнес-логику.
    ТекущийОстаток = ТекущийОстатокПоНН(Источник, НН, ИскомыйТМЦ);
    Если КвоВозврата > ТекущийОстаток + 0.0001 Тогда
        ОшибкаПартии = "Количество возврата по " + Строка(ИскомыйТМЦ)
            + " превышает текущий остаток по НН: " + Строка(ТекущийОстаток);
        Возврат Результат;
    КонецЕсли;

    УжеВозвращено = ИсходноеКво - ТекущийОстаток;
    Если УжеВозвращено < 0 Тогда
        УжеВозвращено = 0;
    КонецЕсли;

    Партии = P281ПрочитатьПартииИзРН(Источник, ИскомыйТМЦ, ОшибкаПартии);
    Если ПустаяСтрока(ОшибкаПартии) = 0 Тогда
        Возврат Результат;
    КонецЕсли;

    ПропуститьБаза = УжеВозвращено * Коэффициент;
    НужноБаза = КвоВозврата * Коэффициент;

    Для Н = 1 По Партии.КоличествоСтрок() Цикл

        Партии.ПолучитьСтрокуПоНомеру(Н);
        ДоступноБаза = Партии.КвоБаза;

        // Сначала "съедаем" партии количеством предыдущих возвратов.
        Если ПропуститьБаза > 0.0001 Тогда
            Если ПропуститьБаза >= ДоступноБаза - 0.0001 Тогда
                ПропуститьБаза = ПропуститьБаза - ДоступноБаза;
                Продолжить;
            Иначе
                ДоступноБаза = ДоступноБаза - ПропуститьБаза;
                ПропуститьБаза = 0;
            КонецЕсли;
        КонецЕсли;

        Если НужноБаза <= 0.0001 Тогда
            Прервать;
        КонецЕсли;

        БеремБаза = НужноБаза;
        Если БеремБаза > ДоступноБаза Тогда
            БеремБаза = ДоступноБаза;
        КонецЕсли;

        Если БеремБаза <= 0.0001 Тогда
            Продолжить;
        КонецЕсли;

        КвоЧасти = БеремБаза / Коэффициент;

        Результат.НоваяСтрока();
        Результат.Партия = Партии.Партия;
        Результат.Кво = КвоЧасти;
        Результат.Коэффициент = Коэффициент;
        Результат.Ед = Единица;
        Результат.ЦенаБезНДС = ЦенаБезНДС;
        Результат.ИсходнаяСуммаБезНДС = ИсходнаяСуммаБезНДС;
        Результат.ИсходныйНДС = ИсходныйНДС;

        НужноБаза = НужноБаза - БеремБаза;

    КонецЦикла;

    Если ПропуститьБаза > 0.0001 Тогда
        ОшибкаПартии = "Предыдущие возвраты превышают количество, списанное по Кт 281 для "
            + Строка(ИскомыйТМЦ) + " в РН " + Источник.НомерДок;
        Возврат Результат;
    КонецЕсли;

    Если НужноБаза > 0.0001 Тогда
        ОшибкаПартии = "Не хватает реальных партий Кт 281 для " + Строка(ИскомыйТМЦ)
            + " в РН " + Источник.НомерДок + ". Нехватка в базовых единицах: " + Строка(НужноБаза);
        Возврат Результат;
    КонецЕсли;

    Возврат Результат;

КонецФункции

Процедура P281ПроверитьПлан(План, ИсточникиПлана, ОшибкаПартии)

    Перем Источник, НН, Распределение;

    ОшибкаПартии = "";

    ИсточникиПлана.ВыбратьСтроки();
    Пока ИсточникиПлана.ПолучитьСтроку() = 1 Цикл

        Источник = СоздатьОбъект("Документ.РасходнаяНакладная");

            ОшибкаПартии = "PARTY281: не удалось открыть НН источника плана";
            Возврат;
        КонецЕсли;

        План.ВыбратьСтроки();
        Пока План.ПолучитьСтроку() = 1 Цикл
            Если План.ИсточникСсылка <> Источник.ТекущийДокумент() Тогда
                Продолжить;

                Источник, НН, План.ТМЦ1С, План.Партия1С, План.Кво, ОшибкаПартии
            );

            Если ПустаяСтрока(ОшибкаПартии) = 0 Тогда
                Возврат;
            КонецЕсли;

            Если Распределение.КоличествоСтрок() = 0 Тогда
                ОшибкаПартии = "PARTY281: пустое распределение для " + Строка(План.ТМЦ1С);
                Возврат;
            КонецЕсли;

        КонецЦикла;

    КонецЦикла;

Функция СоздатьВозвратнуюНакладную(Источник, НН, Товары, НомерВозврата, ДатаВозврата, Магазин)

    Перем Возвратная, ПустаяВозвратная;
    Перем Записано, ПопыткаНомера, ТекстОшибки;
    Перем Распределение, ОшибкаПартии;

    ОшибкаВозвратная = "";
    ПустаяВозвратная = СоздатьОбъект("Документ.ВозвратнаяНакладная");
    Возвратная = СоздатьОбъект("Документ.ВозвратнаяНакладная");
    Возвратная.Новый();
    ЗаполнитьШапкуВозвратной(Возвратная, Источник, НомерВозврата, ДатаВозврата, Магазин);

    Товары.ВыбратьСтроки();
    Пока Товары.ПолучитьСтроку() = 1 Цикл

        // Единственное функциональное изменение рабочего модуля:
        // не переносим Источник.Партия / "Партия по умолчанию".
        // Берем реальные партии из Кт 281 исходной РасходнойНакладной.
        Распределение = P281РаспределитьСтроку(
            Источник, НН, Товары.ТМЦ1С, Товары.Партия1С, Товары.Кво, ОшибкаПартии
        );

        Если ПустаяСтрока(ОшибкаПартии) = 0 Тогда
            ОшибкаВозвратная = ОшибкаПартии;
            Возврат ПустаяВозвратная;
        КонецЕсли;

        Распределение.ВыбратьСтроки();
        Пока Распределение.ПолучитьСтроку() = 1 Цикл

            Возвратная.НоваяСтрока();
            Возвратная.ТМЦ = Товары.ТМЦ1С;
            Возвратная.Партия = Распределение.Партия;
            Возвратная.Кво = Распределение.Кво;
            Возвратная.Коэффициент = Распределение.Коэффициент;
            Возвратная.Ед = Распределение.Ед;
            Возвратная.ЦенаБезНДС = Распределение.ЦенаБезНДС;

            // Формулы продажи оставлены теми же, что были в рабочем модуле.
            Возвратная.СуммаБезНДС = Возвратная.Кво
                * Возвратная.Коэффициент * Возвратная.ЦенаБезНДС;

            Если Распределение.ИсходнаяСуммаБезНДС <> 0 Тогда
                Возвратная.НДС = Возвратная.СуммаБезНДС
                    * Распределение.ИсходныйНДС / Распределение.ИсходнаяСуммаБезНДС;
            Иначе
                Возвратная.НДС = 0;
            КонецЕсли;

            Возвратная.СуммаСНДС = Возвратная.СуммаБезНДС + Возвратная.НДС;
            Возвратная.ДокПродажи = Источник.ТекущийДокумент();

            // Существующая штатная формула себестоимости уже умеет считать
            // по конкретной партии. Теперь ей передается правильная партия Кт 281.
            ЗаполнитьУчетнуюСебестоимостьВозврата(Возвратная, Источник);

        КонецЦикла;

    КонецЦикла;

    Если Возвратная.КоличествоСтрок() = 0 Тогда
        ОшибкаВозвратная = "В ВозвратнаяНакладная не сформировано ни одной строки";
        Возврат ПустаяВозвратная;
    КонецЕсли;

    // Старая рабочая логика предварительной оплаты не меняется.
    ЗаполнитьПредварительнуюОплатуВозврата(Возвратная);

    // Записываем, но не проводим. После проверки бухгалтер проводит документ вручную.
    Записано = 0;
    Для ПопыткаНомера = 1 По 20 Цикл
        Попытка
            Возвратная.Записать();
            Записано = 1;
        Исключение
            ТекстОшибки = ОписаниеОшибки();
            Если (Найти(ТекстОшибки, "Номер не уникальный") > 0) Или (Найти(ТекстОшибки, "унікальн") > 0) Тогда
                Возвратная.УстановитьНовыйНомер("");
            Иначе
                ОшибкаВозвратная = ТекстОшибки;
                Возврат ПустаяВозвратная;
            КонецЕсли;
        КонецПопытки;
        Если Записано = 1 Тогда
            Прервать;
        КонецЕсли;
    КонецЦикла;

    Если Записано = 0 Тогда
        ОшибкаВозвратная = "Не удалось подобрать уникальный номер ВозвратнаяНакладная";
        Возврат ПустаяВозвратная;
    КонецЕсли;

    Если Возвратная.Выбран() = 0 Тогда
        ОшибкаВозвратная = "ВозвратнаяНакладная.Записать() не выбрал созданный документ";
        Возврат ПустаяВозвратная;
    КонецЕсли;

    Возврат Возвратная;

        Если ИсточникСодержитТМЦ(Источник, Запросы.ТМЦ1С) = 1 Тогда
            Количество = Количество + 1;
        КонецЕсли;
    КонецЦикла;
    Возврат Количество;

Функция ЦенаДонораСовпадает(ЦенаСкана, СуммаСНДС, Количество)

    Перем ФактическаяЦенаСНДС, Разница;

    // Если цена со скана отсутствует, не блокируем поиск донора только по цене.
    Если ЦенаСкана <= 0 Тогда
        Возврат 1;
    КонецЕсли;

    Если Количество <= 0 Тогда
        Возврат 0;
    КонецЕсли;

    // Фактическая цена единицы в конкретной РасходнойНакладной.
    // В отличие от ЦенаБезНДС + НДС/Кво, это учитывает скидку и коэффициент.
    ФактическаяЦенаСНДС = СуммаСНДС / Количество;

    Разница = ЦенаСкана - ФактическаяЦенаСНДС;
    Если Разница < 0 Тогда
        Разница = -Разница;
    КонецЕсли;

    Если Разница <= 0.05 Тогда
        Возврат 1;
    КонецЕсли;

    Возврат 0;

// Один проход по РасходнымНакладным на файл (тот же диапазон дат, что в v51).
Процедура КандидатыПостроить(Запросы, ДатаВозврата)

    Перем Н, Фирма, Контрагент, ТоргТочка, ТМЦ, Группа;
    Перем Док, Перв, Кол, ЕстьЗаказ, Заказ;

    КандидатыСбросить();

    // Группы Фирма/Контрагент/ТоргТочка и нужные ТМЦ берем из всех строк файла.
    // Лишний ТМЦ безопасен: он лишь добавит строку, которая не будет выбрана.
    Для Н = 1 По Запросы.КоличествоСтрок() Цикл
        Фирма = Запросы.ПолучитьЗначение(Н, "Фирма");
        Контрагент = Запросы.ПолучитьЗначение(Н, "Контрагент");
        ТоргТочка = Запросы.ПолучитьЗначение(Н, "ТоргТочка");
        ТМЦ = Запросы.ПолучитьЗначение(Н, "ТМЦ1С");
        Если КандидатыНайтиГруппу(Фирма, Контрагент, ТоргТочка) = 0 Тогда
            ГруппыДоноров.НоваяСтрока();
            ГруппыДоноров.Фирма = Фирма;
            ГруппыДоноров.Контрагент = Контрагент;
            ГруппыДоноров.ТоргТочка = ТоргТочка;
        КонецЕсли;
        Если КонтрагентыДоноров.НайтиЗначение(Контрагент) = 0 Тогда
            КонтрагентыДоноров.ДобавитьЗначение(Контрагент);
        КонецЕсли;
        Если НужныеТМЦ.НайтиЗначение(ТМЦ) = 0 Тогда
            НужныеТМЦ.ДобавитьЗначение(ТМЦ);
        КонецЕсли;
    КонецЦикла;

    Док = СоздатьОбъект("Документ.РасходнаяНакладная");
    Док.ВыбратьДокументы('01.01.2000', ДатаВозврата);
    Пока Док.ПолучитьДокумент() = 1 Цикл

        Если Док.Проведен() <> 1 Тогда
            Продолжить;
        КонецЕсли;
        // Быстрый отсев чужих контрагентов, затем точная проверка группы.
        Если КонтрагентыДоноров.НайтиЗначение(Док.Контрагент) = 0 Тогда
            Продолжить;
        КонецЕсли;
        Группа = КандидатыНайтиГруппу(Док.Фирма, Док.Контрагент, Док.ТоргТочка);
        Если Группа = 0 Тогда
            Продолжить;
        КонецЕсли;

        Перв = КандСтроки.КоличествоСтрок() + 1;
        Кол = 0;
        Док.ВыбратьСтроки();
        Пока Док.ПолучитьСтроку() = 1 Цикл
            Если НужныеТМЦ.НайтиЗначение(Док.ТМЦ) > 0 Тогда
                КандСтроки.НоваяСтрока();
                КандСтроки.ТМЦ = Док.ТМЦ;
                КандСтроки.Партия = Док.Партия;
                КандСтроки.Кво = Док.Кво;
                КандСтроки.ЦенаБезНДС = Док.ЦенаБезНДС;
                КандСтроки.НДС = Док.НДС;
                КандСтроки.СуммаСНДС = Док.СуммаСНДС;
                Кол = Кол + 1;
            КонецЕсли;

        Если Кол = 0 Тогда

        КонецЕсли;

        Заказ = 0;

            Если Док.Заказ.Выбран() = 1 Тогда
                Заказ = Док.Заказ;
                ЕстьЗаказ = 1;
            КонецЕсли;
        Исключение
        КонецПопытки;

        Кандидаты.НоваяСтрока();
        Кандидаты.Группа = Группа;
        Кандидаты.Ссылка = Док.ТекущийДокумент();
        Кандидаты.ДатаРН = Док.ДатаДок;
        Кандидаты.ЕстьЗаказ = ЕстьЗаказ;
        Кандидаты.Заказ = Заказ;
        Кандидаты.Перв = Перв;
        Кандидаты.Кол = Кол;
        Кандидаты.ННСтатус = 0;
        Кандидаты.ННСсылка = 0;
        Кандидаты.ННФирма = 0;
        Кандидаты.ННКонтрагент = 0;
        Кандидаты.ННПерв = 0;
        Кандидаты.ННКол = 0;

// Та же логика, что ПолучитьДанныеСтрокиИсточника, но по запомненным строкам.

    Если КандСтрокиНН.ПолучитьЗначение(СтрокаНН, "Версия") = ИндексП2Версия Тогда
        Возврат КандСтрокиНН.ПолучитьЗначение(СтрокаНН, "Остаток");
    КонецЕсли;

    НомерСтрокиНН = КандСтрокиНН.ПолучитьЗначение(СтрокаНН, "НомерСтроки");
    КвоНН = КандСтрокиНН.ПолучитьЗначение(СтрокаНН, "Кво");
    ТекНомер = НомерСтрокиНН;
    ТекКво = КвоНН;
    Было = 0;

    СостояниеП2Рассчитать(
        Кандидаты.ПолучитьЗначение(НК, "Ссылка"),
        Кандидаты.ПолучитьЗначение(НК, "ЕстьЗаказ"),
        Кандидаты.ПолучитьЗначение(НК, "Заказ"),
        ННСсылка,
        Кандидаты.ПолучитьЗначение(НК, "ННФирма"),
        ННКонтрагент,
        ИскомыйТМЦ, НомерСтрокиНН, КвоНН, ТекНомер, ТекКво, Было
    );

    Если ТекКво < 0 Тогда
        ТекКво = 0;
    КонецЕсли;

    КандСтрокиНН.УстановитьЗначение(СтрокаНН, "Остаток", ТекКво);
    КандСтрокиНН.УстановитьЗначение(СтрокаНН, "Версия", ИндексП2Версия);

    Возврат ТекКво;

Процедура СформироватьПланДоноров(Запросы, ДатаВозврата, План, ИсточникиПлана, ОшибкаПлана)

    Перем Источник, НН, ЛучшийТМЦ, ОшибкаПоиска;
    Перем Доступно, КвоВПлан, НашлиСтроку, ПартияСтроки;
    Перем КвоСтроки, ЦенаБезНДССтроки, НДССтроки, СуммаСНДССтроки, ЦенаОценка;
    Перем Найдено, ЛучшаяСсылка, ЛучшаяННСсылка, ЛучшееДоступно;
    Перем ЕстьОстаток, БылоРаспределение, Шаг, ЗапросыОценки;
    Перем ИсточникДонор, ННДонор;
    Перем ЕстьПредПартия, ПредКодПартии, ПредДатаПартии;

    ОшибкаПлана = "";
    КандидатыСбросить();

    // ЭТАП 1а: для каждой строки находим родную РН, ее НН и точный ТМЦ.
    // Проверки, их порядок и тексты ошибок — как в этапе 1 v51.
    // Здесь ничего не распределяется, поэтому разделение этапа 1 на 1а/1б
    // не меняет ни результат, ни первую найденную ошибку.
    ЕстьПредПартия = 0;
    ПредКодПартии = "";
    ПредДатаПартии = ПолучитьПустоеЗначение("Дата");

    Запросы.ВыбратьСтроки();
    Пока Запросы.ПолучитьСтроку() = 1 Цикл
        Если (ПустаяСтрока(Запросы.КодПартии) = 1) Или (ПустоеЗначение(Запросы.ДатаПартии) = 1) Тогда
            ОшибкаПлана = "Не указаны номер или дата партии для " + Запросы.Наименование;
            Возврат;
        КонецЕсли;
        Если (ПустаяСтрока(Запросы.Наименование) = 1) Или (Запросы.Кво <= 0) Тогда
            ОшибкаПлана = "В строке импорта отсутствует товар или количество";
            Возврат;
        КонецЕсли;

        // Товары одного BATCH идут подряд и относятся к одной РН.
        // Она уже прошла все проверки ниже, поэтому повторно ее не ищем.
        Если ЕстьПредПартия = 1 Тогда
            Если (Запросы.КодПартии <> ПредКодПартии) Или (Запросы.ДатаПартии <> ПредДатаПартии) Тогда
                ЕстьПредПартия = 0;
            КонецЕсли;
        КонецЕсли;

        Если ЕстьПредПартия = 0 Тогда
            Источник = СоздатьОбъект("Документ.РасходнаяНакладная");
            Если НайтиРасходнуюНакладную(Источник, Запросы.КодПартии, Запросы.ДатаПартии) = 0 Тогда
                ОшибкаПлана = "Не найдена исходная РасходнаяНакладная № " + Запросы.КодПартии + " от " + Строка(Запросы.ДатаПартии);
                Возврат;
            КонецЕсли;
            Если Источник.Проведен() <> 1 Тогда
                ОшибкаПлана = "Исходная РасходнаяНакладная № " + Источник.НомерДок + " не проведена";
                Возврат;
            КонецЕсли;

            НН = СоздатьОбъект("Документ.НалоговаяНакладная");
            Если НайтиНалоговуюНакладную(Источник, НН) = 0 Тогда
                ОшибкаПлана = "У исходной РасходнаяНакладная № " + Источник.НомерДок + " нет подчиненной НалоговаяНакладная";
                Возврат;
            КонецЕсли;

            ЕстьПредПартия = 1;
            ПредКодПартии = Запросы.КодПартии;
            ПредДатаПартии = Запросы.ДатаПартии;

            // Остатки по этой НН понадобятся на этапе 1б: индекс П2
            // будет построен один раз сразу для всех родных НН файла.
            ИндексП2ДобавитьПокрытие(НН.ТекущийДокумент(), НН.Контрагент);
        КонецЕсли;

        ОпределитьТМЦВИсточнике(Источник, НН, Запросы.Наименование, Запросы.ЦенаСкана, ЛучшийТМЦ, ОшибкаПоиска);
        Если ПустаяСтрока(ОшибкаПоиска) = 0 Тогда
            ОшибкаПлана = ОшибкаПоиска;
            Возврат;
        КонецЕсли;

        Запросы.ТМЦ1С = ЛучшийТМЦ;
        Запросы.ИсточникСсылка = Источник.ТекущийДокумент();
        Запросы.ННСсылка = НН.ТекущийДокумент();
        Запросы.Фирма = Источник.Фирма;
        Запросы.Контрагент = Источник.Контрагент;
        Запросы.ТоргТочка = Источник.ТоргТочка;
        Запросы.Осталось = Запросы.Кво;

        ПолучитьДанныеСтрокиИсточника(Источник, ЛучшийТМЦ, Запросы.ЦенаСкана, НашлиСтроку, ПартияСтроки, КвоСтроки, ЦенаБезНДССтроки, НДССтроки, СуммаСНДССтроки, ЦенаОценка);
        Если НашлиСтроку = 0 Тогда
            ОшибкаПлана = "Не найдена точная строка ТМЦ в исходной РасходнаяНакладная: " + Запросы.Наименование;
            Возврат;
        КонецЕсли;
        Запросы.Партия1С = ПартияСтроки;
    КонецЦикла;

    // ЭТАП 1б: доступный остаток родной расходной (как в конце этапа 1 v51,
    // в том же порядке строк).
    Запросы.ВыбратьСтроки();
    Пока Запросы.ПолучитьСтроку() = 1 Цикл
        Источник = СоздатьОбъект("Документ.РасходнаяНакладная");
        Источник.НайтиДокумент(Запросы.ИсточникСсылка);
        НН = СоздатьОбъект("Документ.НалоговаяНакладная");
        НН.НайтиДокумент(Запросы.ННСсылка);

        Доступно = ДоступноДляПлана(Источник, НН, Запросы.ТМЦ1С, План);
        Если Доступно > 0 Тогда
            КвоВПлан = Запросы.Осталось;
            Если КвоВПлан > Доступно Тогда
                КвоВПлан = Доступно;
            КонецЕсли;
            ДобавитьВПлан(План, ИсточникиПлана, Запросы.ИсточникСсылка, Запросы.ННСсылка, Запросы.ТМЦ1С, Запросы.Партия1С, Запросы.Наименование, Запросы.ЦенаСкана, КвоВПлан);
            Запросы.Осталось = Запросы.Осталось - КвоВПлан;
        КонецЕсли;
    КонецЦикла;

    // ЭТАП 2: остаток раскладываем по донорам.
    Для Шаг = 1 По 10000 Цикл
        ЕстьОстаток = 0;
        БылоРаспределение = 0;

        // Снимок защищает курсор Запросы от вложенной оценки кандидатов.
        ЗапросыОценки = КопияЗапросовДляОценки(Запросы);
        Запросы.ВыбратьСтроки();
        Пока Запросы.ПолучитьСтроку() = 1 Цикл
            Если Запросы.Осталось <= 0 Тогда
                Продолжить;
            КонецЕсли;
            ЕстьОстаток = 1;

            НайтиИспользованныйИсточник(План, Запросы.Фирма, Запросы.Контрагент, Запросы.ТоргТочка, Запросы.ТМЦ1С, Запросы.ЦенаСкана, Запросы.Осталось, Найдено, ЛучшаяСсылка, ЛучшаяННСсылка, ЛучшееДоступно);
            Если Найдено = 0 Тогда
                НайтиЛучшегоДонора(ЗапросыОценки, План, ДатаВозврата, Запросы.Фирма, Запросы.Контрагент, Запросы.ТоргТочка, Запросы.ТМЦ1С, Запросы.ЦенаСкана, Запросы.Осталось, Найдено, ЛучшаяСсылка, ЛучшаяННСсылка, ЛучшееДоступно);
            КонецЕсли;

            Если Найдено = 0 Тогда
                ОшибкаПлана = "Не найден донор с остатком и подходящей ценой для " + Запросы.Наименование + ". Нехватка: " + Строка(Запросы.Осталось) + ". Требуется тот же Контрагент и та же ТоргТочка.";
                Возврат;
            КонецЕсли;

            ИсточникДонор = СоздатьОбъект("Документ.РасходнаяНакладная");
            Если ИсточникДонор.НайтиДокумент(ЛучшаяСсылка) = 0 Тогда
                ОшибкаПлана = "Планировщик потерял выбранную донорскую РасходнаяНакладная";
                Возврат;
            КонецЕсли;
            ННДонор = СоздатьОбъект("Документ.НалоговаяНакладная");
            Если ННДонор.НайтиДокумент(ЛучшаяННСсылка) = 0 Тогда
                ОшибкаПлана = "Планировщик потерял НалоговаяНакладная донора";
                Возврат;
            КонецЕсли;

            ПолучитьДанныеСтрокиИсточника(ИсточникДонор, Запросы.ТМЦ1С, Запросы.ЦенаСкана, НашлиСтроку, ПартияСтроки, КвоСтроки, ЦенаБезНДССтроки, НДССтроки, СуммаСНДССтроки, ЦенаОценка);
            Если НашлиСтроку = 0 Тогда
                ОшибкаПлана = "В выбранном доноре пропала строка товара " + Запросы.Наименование;
                Возврат;
            КонецЕсли;

            Доступно = ДоступноДляПлана(ИсточникДонор, ННДонор, Запросы.ТМЦ1С, План);
            Если Доступно <= 0 Тогда
                ОшибкаПлана = "У выбранного донора закончился плановый остаток по " + Запросы.Наименование;
                Возврат;
            КонецЕсли;

            КвоВПлан = Запросы.Осталось;
            Если КвоВПлан > Доступно Тогда
                КвоВПлан = Доступно;
            КонецЕсли;

            ДобавитьВПлан(План, ИсточникиПлана, ИсточникДонор.ТекущийДокумент(), ННДонор.ТекущийДокумент(), Запросы.ТМЦ1С, ПартияСтроки, Запросы.Наименование, Запросы.ЦенаСкана, КвоВПлан);
            Запросы.Осталось = Запросы.Осталось - КвоВПлан;
            БылоРаспределение = 1;
        КонецЦикла;

        Если ЕстьОстаток = 0 Тогда
            Прервать;
        КонецЕсли;
        Если БылоРаспределение = 0 Тогда
            ОшибкаПлана = "Планировщик доноров не смог распределить оставшиеся товары";
            Возврат;
        КонецЕсли;
    КонецЦикла;

    Запросы.ВыбратьСтроки();
    Пока Запросы.ПолучитьСтроку() = 1 Цикл
        Если Запросы.Осталось > 0 Тогда
            ОшибкаПлана = "Не удалось полностью распределить " + Запросы.Наименование + ". Осталось: " + Строка(Запросы.Осталось);
            Возврат;
        КонецЕсли;
    КонецЦикла;

    Если План.КоличествоСтрок() = 0 Тогда
        ОшибкаПлана = "План возврата пуст";
    КонецЕсли;

Функция ОписаниеПланаДоноров(План, ИсточникиПлана)

    Перем Результат, Источник, ТоварыВИсточнике;

Функция ВыполнитьПланДоноров(План, ИсточникиПлана, НомерВозврата, ДатаВозврата, Магазин)

    Перем Источник, НН, Товары, Сообщение, Результат;
    Результат = "";

    ИсточникиПлана.ВыбратьСтроки();
    Пока ИсточникиПлана.ПолучитьСтроку() = 1 Цикл
        Источник = СоздатьОбъект("Документ.РасходнаяНакладная");
        Если Источник.НайтиДокумент(ИсточникиПлана.ИсточникСсылка) = 0 Тогда
            Возврат "ERROR|Не удалось открыть источник из готового плана";
        КонецЕсли;
        НН = СоздатьОбъект("Документ.НалоговаяНакладная");
        Если НН.НайтиДокумент(ИсточникиПлана.ННСсылка) = 0 Тогда
            Возврат "ERROR|Не удалось открыть НалоговаяНакладная из готового плана";
        КонецЕсли;

        Товары = НоваяТаблицаТоваров();
        План.ВыбратьСтроки();
        Пока План.ПолучитьСтроку() = 1 Цикл
            Если План.ИсточникСсылка = Источник.ТекущийДокумент() Тогда
                Товары.НоваяСтрока();
                Товары.Наименование = План.Наименование;
                Товары.Кво = План.Кво;
                Товары.ЦенаСкана = План.ЦенаСкана;
                Товары.ТМЦ1С = План.ТМЦ1С;
                Товары.Партия1С = План.Партия1С;
            КонецЕсли;
        КонецЦикла;

        Если Товары.КоличествоСтрок() = 0 Тогда
            Возврат "ERROR|Для источника " + Источник.НомерДок + " в плане нет товаров";
        КонецЕсли;

        Сообщение = ИмпортироватьПоИсточникуПлана(Источник, НН, Товары, НомерВозврата, ДатаВозврата, Магазин);
        Если Лев(Сообщение, 6) = "ERROR|" Тогда
            Возврат Сообщение;
        КонецЕсли;

        Если ПустаяСтрока(Результат) = 1 Тогда
            Результат = Сред(Сообщение, 4, СтрДлина(Сообщение) - 3);
        Иначе
            Результат = Результат + "; " + Сред(Сообщение, 4, СтрДлина(Сообщение) - 3);
        КонецЕсли;
    КонецЦикла;

    Возврат "OK|" + Результат;

Функция ИмпортироватьПакет(КодПартии, ДатаПартии, Товары, НомерВозврата, ДатаВозврата, Магазин)

    Перем Источник, НН, Сообщение, СуществующийВозврат;
    Перем Возвратная, Корректировка;

    Если (ПустаяСтрока(КодПартии) = 1) Или (ПустоеЗначение(ДатаПартии) = 1) Тогда
        Возврат "ERROR|Не указаны номер или дата партии";
    КонецЕсли;
    Если Товары.КоличествоСтрок() = 0 Тогда
        Возврат "ERROR|В блоке " + КодПартии + " нет товаров";
    КонецЕсли;
    Если ПустоеЗначение(ДатаВозврата) = 1 Тогда
        Возврат "ERROR|Не указана дата накладной на возврат";
    КонецЕсли;
    Если ПустаяСтрока(СокрЛП(Магазин)) = 1 Тогда
        Возврат "ERROR|Не распознан номер/адрес торговой точки (STORE). Документы не созданы.";
    КонецЕсли;

    Источник = СоздатьОбъект("Документ.РасходнаяНакладная");
    Если НайтиРасходнуюНакладную(Источник, КодПартии, ДатаПартии) = 0 Тогда
        Возврат "ERROR|Не найдена РасходнаяНакладная № " + КодПартии + " от " + Строка(ДатаПартии);
    КонецЕсли;

    НН = СоздатьОбъект("Документ.НалоговаяНакладная");
    Если НайтиНалоговуюНакладную(Источник, НН) = 0 Тогда
        Возврат "ERROR|У расходной накладной № " + Источник.НомерДок + " нет подчиненной НалоговаяНакладная";
    КонецЕсли;

    Сообщение = ПроверитьТовары(Источник, НН, Товары);
    Если ПустаяСтрока(Сообщение) = 0 Тогда
        Возврат "ERROR|" + Сообщение;
    КонецЕсли;

    СуществующийВозврат = НайтиУжеИмпортированныйВозврат(Источник, НомерВозврата, ДатаВозврата);
    Если СуществующийВозврат.Выбран() = 1 Тогда
        Если ТолькоПроверка = 1 Тогда
            Возврат "OK|ПРОВЕРКА: уже существует возврат № " + СуществующийВозврат.НомерДок;
        КонецЕсли;

        // LIVE SAFE: проведенный документ никогда не изменяем.
        Если СуществующийВозврат.Проведен() = 1 Тогда
            Возврат "ERROR|ВозвратнаяНакладная № " + СуществующийВозврат.НомерДок + " уже проведена. AutoImport запрещено изменять проведенные документы.";
        КонецЕсли;

        // Старые версии AutoImport могли записать возврат без Фирмы/Заказа.
        // Дозаполняем шапку, чтобы документ был виден в журнале и готов к проверке.
        Попытка
            ЗаполнитьШапкуВозвратной(СуществующийВозврат, Источник, НомерВозврата, ДатаВозврата, Магазин);
            ЗаполнитьПредварительнуюОплатуВозврата(СуществующийВозврат);
            СуществующийВозврат.Записать();
        Исключение
            Возврат "ERROR|Найден существующий возврат, но не удалось исправить его реквизиты: " + ОписаниеОшибки();
        КонецПопытки;

        Корректировка = НайтиКорректировкуНДС(Источник, СуществующийВозврат);
        Если Корректировка.Выбран() = 0 Тогда
            Корректировка = СоздатьКорректировкуНДС(Источник, НН, СуществующийВозврат, ДатаВозврата);
        КонецЕсли;

        Если ЭкспортироватьВозвратВMXL(СуществующийВозврат) = 0 Тогда
            Возврат "ERROR|ВозвратнаяНакладная " + СуществующийВозврат.НомерДок + " записана, но файл RETURN_* не создан";
        КонецЕсли;
        Если Корректировка.Выбран() = 0 Тогда
            Возврат "ERROR|ВозвратнаяНакладная " + СуществующийВозврат.НомерДок + " существует, но Приложение2 не записано: " + ОшибкаПриложение2;
        КонецЕсли;

        Если ЭкспортироватьПриложение2ВMXL(Корректировка) = 0 Тогда
            Возврат "ERROR|Приложение2 " + Корректировка.НомерДок + " записано, но файл P2_* не создан";
        КонецЕсли;
        Возврат "OK|Уже существует возврат № " + СуществующийВозврат.НомерДок + "; Приложение2 " + Корректировка.НомерДок;
    КонецЕсли;

    Если ТолькоПроверка = 1 Тогда
        Возврат "OK|ПРОВЕРКА: Расх. накл. " + Источник.НомерДок + "; НН " + НН.НомерДок + "; строк: " + Строка(Товары.КоличествоСтрок());
    КонецЕсли;

    Возвратная = СоздатьВозвратнуюНакладную(Источник, НН, Товары, НомерВозврата, ДатаВозврата, Магазин);
    Если Возвратная.Выбран() = 0 Тогда
        Возврат "ERROR|Не удалось записать ВозвратнаяНакладная: " + ОшибкаВозвратная;
    КонецЕсли;

    Корректировка = СоздатьКорректировкуНДС(Источник, НН, Возвратная, ДатаВозврата);
    Если Корректировка.Выбран() = 0 Тогда
        Если ЭкспортироватьВозвратВMXL(Возвратная) = 0 Тогда
            Возврат "ERROR|ВозвратнаяНакладная " + Возвратная.НомерДок + " создана, но не созданы Приложение2 и файл RETURN_*: " + ОшибкаПриложение2;
        КонецЕсли;
        Возврат "ERROR|ВозвратнаяНакладная " + Возвратная.НомерДок + " создана, но Приложение2 не записано: " + ОшибкаПриложение2;
    КонецЕсли;

    Если ЭкспортироватьВозвратВMXL(Возвратная) = 0 Тогда
        Возврат "ERROR|ВозвратнаяНакладная " + Возвратная.НомерДок + " записана, но файл RETURN_* не создан";
    КонецЕсли;
    Если ЭкспортироватьПриложение2ВMXL(Корректировка) = 0 Тогда
        Возврат "ERROR|Приложение2 " + Корректировка.НомерДок + " записано, но файл P2_* не создан";
    КонецЕсли;

    Возврат "OK|ВозвратнаяНакладная " + Возвратная.НомерДок + "; Приложение2 " + Корректировка.НомерДок;

Функция ОбработатьФайлЗадания(ПолноеИмя)

    Перем ТекстЗадания, НомерСтроки, СтрокаФайла, ТипСтроки;
    Перем ИдЗадания, НомерВозврата, ДатаВозврата, Магазин;
    Перем КодПартии, ДатаПартии, Запросы, План, ИсточникиПлана;
    Перем ПозицийВБлоке, ОшибкаПлана, Сообщение, ОписаниеПлана;

    ТекстЗадания = СоздатьОбъект("Текст");
    ТекстЗадания.Открыть(ПолноеИмя);

    ИдЗадания = "";
    НомерВозврата = "";
    ДатаВозврата = ПолучитьПустоеЗначение("Дата");
    Магазин = "";
    КодПартии = "";
    ДатаПартии = ПолучитьПустоеЗначение("Дата");
    ПозицийВБлоке = 0;

    Запросы = НоваяТаблицаЗапросовДоноров();
    План = НоваяТаблицаПланаДоноров();
    ИсточникиПлана = НоваяТаблицаИсточниковПлана();

    // v50: сначала читаем ВЕСЬ файл. На этапе разбора ничего в 1С не создается.
    Для НомерСтроки = 1 По ТекстЗадания.КоличествоСтрок() Цикл
        СтрокаФайла = ТекстЗадания.ПолучитьСтроку(НомерСтроки);
        Если ПустаяСтрока(СтрокаФайла) = 1 Тогда
            Продолжить;
        КонецЕсли;
        ТипСтроки = ПолеФайла(СтрокаФайла, 1);

        Если ТипСтроки = "FORMAT" Тогда
            Если ПолеФайла(СтрокаФайла, 2) <> "3" Тогда
                Возврат "ERROR|Неподдерживаемая версия файла";
            КонецЕсли;
        ИначеЕсли ТипСтроки = "TASK_ID" Тогда
            ИдЗадания = ПолеФайла(СтрокаФайла, 2);
        ИначеЕсли ТипСтроки = "DOC_NUM" Тогда
            НомерВозврата = ПолеФайла(СтрокаФайла, 2);
        ИначеЕсли ТипСтроки = "DATE" Тогда
            ДатаВозврата = ДатаИзДДММГГ(ПолеФайла(СтрокаФайла, 2));
        ИначеЕсли ТипСтроки = "STORE" Тогда
            Магазин = ПолеФайла(СтрокаФайла, 2);
        ИначеЕсли ТипСтроки = "BATCH" Тогда
            Если ПустаяСтрока(КодПартии) = 0 Тогда
                Возврат "ERROR|В файле отсутствует END_BATCH перед следующим BATCH";
            КонецЕсли;
            КодПартии = ПолеФайла(СтрокаФайла, 2);
            ДатаПартии = ДатаИзДДММГГ(ПолеФайла(СтрокаФайла, 3));
            ПозицийВБлоке = 0;
        ИначеЕсли ТипСтроки = "ITEM" Тогда
            Если ПустаяСтрока(КодПартии) = 1 Тогда
                Возврат "ERROR|Строка ITEM указана вне блока BATCH";
            КонецЕсли;
            Запросы.НоваяСтрока();
            Запросы.КодПартии = КодПартии;
            Запросы.ДатаПартии = ДатаПартии;
            Запросы.Наименование = ПолеФайла(СтрокаФайла, 2);
            Запросы.Кво = ЧислоИзФайла(ПолеФайла(СтрокаФайла, 3));
            Запросы.ЦенаСкана = ЧислоИзФайла(ПолеФайла(СтрокаФайла, 4));
            Запросы.Осталось = Запросы.Кво;
            ПозицийВБлоке = ПозицийВБлоке + 1;
        ИначеЕсли ТипСтроки = "END_BATCH" Тогда
            Если ПустаяСтрока(КодПартии) = 1 Тогда
                Возврат "ERROR|END_BATCH указан без BATCH";
            КонецЕсли;
            Если ПозицийВБлоке = 0 Тогда
                Возврат "ERROR|В блоке " + КодПартии + " нет товаров";
            КонецЕсли;
            КодПартии = "";
            ДатаПартии = ПолучитьПустоеЗначение("Дата");
            ПозицийВБлоке = 0;
        КонецЕсли;
    КонецЦикла;

    Если ПустаяСтрока(ИдЗадания) = 1 Тогда
        Возврат "ERROR|В файле нет TASK_ID";
    КонецЕсли;
    Если ПустаяСтрока(КодПартии) = 0 Тогда
        Возврат "ERROR|Файл заканчивается внутри блока BATCH";
    КонецЕсли;
    Если Запросы.КоличествоСтрок() = 0 Тогда
        Возврат "ERROR|В файле нет ни одного товара";
    КонецЕсли;
    Если ПустоеЗначение(ДатаВозврата) = 1 Тогда
        Возврат "ERROR|Не указана дата накладной на возврат";
    КонецЕсли;
    Если ПустаяСтрока(СокрЛП(Магазин)) = 1 Тогда
        Возврат "ERROR|Не распознан номер/адрес торговой точки (STORE). Документы не созданы.";
    КонецЕсли;

    // После успешного донорского импорта остатки изменяются. Поэтому повторное
    // перепланирование того же внешнего номера запрещаем, чтобы не уйти на новых
    // доноров и не создать дубль.
    Если НайтиСуществующийВозвратПоВнешнемуНомеру(НомерВозврата, ДатаВозврата, Магазин) = 1 Тогда
        Возврат "ERROR|По номеру возврата " + НомерВозврата + " от " + Строка(ДатаВозврата) + " уже существует ВозвратнаяНакладная. Повторное донорское планирование заблокировано; сначала проверьте существующий документ.";
    КонецЕсли;

    // Ключевая гарантия: до завершения СформироватьПланДоноров документы не пишутся.
    СформироватьПланДоноров(Запросы, ДатаВозврата, План, ИсточникиПлана, ОшибкаПлана);
    Если ПустаяСтрока(ОшибкаПлана) = 0 Тогда
        Возврат "ERROR|" + ОшибкаПлана + " Документы не созданы.";
    КонецЕсли;

    // PARTY281 preflight: проверяем реальные партии всех источников до записи
    // первого документа, чтобы сохранить старую гарантию "без частичного создания".
    P281ПроверитьПлан(План, ИсточникиПлана, ОшибкаПлана);
    Если ПустаяСтрока(ОшибкаПлана) = 0 Тогда
        Возврат "ERROR|PARTY281: " + ОшибкаПлана + " Документы не созданы.";
    КонецЕсли;

    ОписаниеПлана = ОписаниеПланаДоноров(План, ИсточникиПлана);

    Если ТолькоПроверка = 1 Тогда
        ЗаписатьСтатусРезультата(ИдЗадания, "ПРОВЕРКА DONOR PLAN: " + ОписаниеПлана);
        Возврат "OK|" + ИдЗадания + "|ПРОВЕРКА DONOR PLAN: " + ОписаниеПлана;
    КонецЕсли;

    Сообщение = ВыполнитьПланДоноров(План, ИсточникиПлана, НомерВозврата, ДатаВозврата, Магазин);
    Если Лев(Сообщение, 6) = "ERROR|" Тогда
        Возврат Сообщение;
    КонецЕсли;

    ЗаписатьСтатусРезультата(ИдЗадания, "План: " + ОписаниеПлана + "; " + Сред(Сообщение, 4, СтрДлина(Сообщение) - 3));
    Возврат "OK|" + ИдЗадания + "|План: " + ОписаниеПлана + "; " + Сред(Сообщение, 4, СтрДлина(Сообщение) - 3);

КонецФункции

// LIVE SAFE: не запускаем боевой мониторинг поверх старых *.ready.
// 1 = старт разрешен; 0 = старт заблокирован.
Функция БезопасныйСтартРазрешен()

    Перем Входящие, ИмяФайла;

    Входящие = ПолучитьКаталогОбмена() + "\\inbox";

    Попытка
        ФС.УстТекКаталог(Входящие);
        ИмяФайла = ФС.НайтиПервыйФайл("*.ready");
    Исключение
        Сообщить("AutoImport v37 LIVE SAFE: не удалось проверить inbox: " + ОписаниеОшибки(), "!");
        Возврат 0;
    КонецПопытки;

    Если ПустаяСтрока(ИмяФайла) = 0 Тогда
        Сообщить("AutoImport v37 LIVE SAFE: запуск заблокирован. В inbox уже есть " + ИмяФайла + ". Уберите старые *.ready и откройте обработку заново.", "!");
        Возврат 0;
    КонецЕсли;

    Возврат 1;

// v51: после любого результата проверяем фактическое перемещение файла.
Функция ПеренестиФайлЗадания(ПолноеИмя, КаталогНазначения, ИмяНазначения, ОшибкаПереноса)

    Перем Файлы;

    ОшибкаПереноса = "";
    Файлы = СоздатьОбъект("ФС");

    // Создание уже существующего каталога не должно мешать переносу.
    // Если каталог создать нельзя, ошибка самого переноса будет показана ниже.
    Попытка
        Файлы.СоздатьКаталог(КаталогНазначения);
    Исключение
        ОшибкаПереноса = "";
    КонецПопытки;

    Попытка
        Файлы.ПереименоватьФайл(ПолноеИмя, ИмяНазначения, 1);
    Исключение
        ОшибкаПереноса = ОписаниеОшибки();
    КонецПопытки;

    // У ПереименоватьФайл нет кода возврата: проверяем оба пути.
    Попытка
        Если (Файлы.СуществуетФайл(ПолноеИмя) = 0)
           И (Файлы.СуществуетФайл(ИмяНазначения) = 1) Тогда
            Возврат 1;
        КонецЕсли;
    Исключение
        ОшибкаПереноса = ОшибкаПереноса + " " + ОписаниеОшибки();
    КонецПопытки;

    Если ПустаяСтрока(ОшибкаПереноса) = 1 Тогда
        ОшибкаПереноса = "Файл остался в inbox или не появился в папке назначения. Проверьте доступ к папкам и блокировку файла.";
    КонецЕсли;

    Возврат 0;

Процедура ПроверитьВходящие()

    Перем Входящие, ИмяФайла, ПолноеИмя, Сообщение, ИмяНазначения;
    Перем Файлы, СписокФайлов, НомерФайла, КаталогНазначения, ОшибкаПереноса;

    Если (СейчасОбрабатываем = 1) Или (ИмпортОстановлен = 1) Тогда

    КонецЕсли;

    СейчасОбрабатываем = 1;
    Входящие = ПолучитьКаталогОбмена() + "\inbox";

    // Собираем очередь до обработки: вызовы из документов/печати не собьют
    // выборку файлов. Способ установки каталога сохранен из рабочего модуля.
    Файлы = СоздатьОбъект("ФС");
    СписокФайлов = СоздатьОбъект("ТаблицаЗначений");
    СписокФайлов.НоваяКолонка("ИмяФайла");
    Файлы.УстТекКаталог(Входящие);
    ИмяФайла = Файлы.НайтиПервыйФайл("*.ready");
    Пока ПустаяСтрока(ИмяФайла) = 0 Цикл
        СписокФайлов.НоваяСтрока();
        СписокФайлов.ИмяФайла = ИмяФайла;
        ИмяФайла = Файлы.НайтиСледующийФайл();
    КонецЦикла;

    // Если файлов нет, молча ждем следующего такта таймера.
    Для НомерФайла = 1 По СписокФайлов.КоличествоСтрок() Цикл
        СписокФайлов.ПолучитьСтрокуПоНомеру(НомерФайла);
        ИмяФайла = СписокФайлов.ИмяФайла;
        ПолноеИмя = Входящие + "\" + ИмяФайла;
        Если Файлы.СуществуетФайл(ПолноеИмя) = 0 Тогда
            Продолжить;
        КонецЕсли;

        // Кэши действуют только для одного файла.
        СброситьКэшиФайла();
        КэшСостоянияП2Активен = 1;
        Попытка
            Сообщение = ОбработатьФайлЗадания(ПолноеИмя);
        Исключение
            Сообщение = "ERROR|Исключение при обработке " + ИмяФайла + ": " + ОписаниеОшибки();
        КонецПопытки;
        КэшСостоянияП2Активен = 0;
        СброситьКэшиФайла();

        Если Лев(Сообщение, 3) = "OK|" Тогда
            КаталогНазначения = ПолучитьКаталогОбмена() + "\processed";
            ИмяНазначения = КаталогНазначения + "\" + ИмяФайла + ".done";
        Иначе
            // Даже ошибка записи журнала не должна оставлять задание
            // для повторного импорта на следующем такте таймера.
            Попытка

            Исключение
                Сообщить("AutoImport: не удалось записать журнал ошибки: " + Лев(ОписаниеОшибки(), 160), "!");
            КонецПопытки;

            ИмяНазначения = КаталогНазначения + "\" + ИмяФайла + ".error";
        КонецЕсли;

        Если ПеренестиФайлЗадания(ПолноеИмя, КаталогНазначения, ИмяНазначения, ОшибкаПереноса) = 0 Тогда
            // Если Windows не разрешила перенос, повторять создание документов
            // нельзя. Возобновление — после устранения причины и открытия формы.
            ИмпортОстановлен = 1;
            Форма.ОбработкаОжидания("", 0);
            СейчасОбрабатываем = 0;
            Сообщить("AutoImport: мониторинг остановлен — файл не перемещен. Повторный импорт заблокирован.", "!");
            Сообщить("Источник: " + Лев(ПолноеИмя, 220), "!");
            Сообщить("Назначение: " + Лев(ИмяНазначения, 220), "!");
            Сообщить(Лев(ОшибкаПереноса, 240), "!");
            Сообщить("Результат обработки: " + Лев(Сообщение, 210), "!");
            Возврат;
        КонецЕсли;

        Если Лев(Сообщение, 3) = "OK|" Тогда
            Сообщить("AutoImport: " + ИмяФайла + " оброблено. Деталі записані у папку result.", "i");
        Иначе
            Если СтрДлина(Сообщение) > 240 Тогда
                Сообщить(Лев(Сообщение, 240) + "... Деталі записані у папку result.", "!");
            Иначе
                Сообщить(Сообщение, "!");
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;

    СейчасОбрабатываем = 0;
